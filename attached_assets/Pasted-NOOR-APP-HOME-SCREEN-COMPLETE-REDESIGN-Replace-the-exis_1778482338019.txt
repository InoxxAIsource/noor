import { useState, useEffect, useRef } from 'react'
import { useNavigate } from 'react-router-dom'
import useAuth from '../hooks/useAuth'

export default function Home() {
  const { user } = useAuth()
  const navigate = useNavigate()
  const [prayerTimes, setPrayerTimes] = useState(null)
  const [nextPrayer, setNextPrayer] = useState({ name:'Asr', time:'15:58' })
  const [countdown, setCountdown] = useState('--:--:--')
  const [iftarCountdown, setIftarCountdown] = useState('--:--:--')
  const [streak, setStreak] = useState({ current:0, weekly:0, weeklyGoal:5 })
  const [hijriDate, setHijriDate] = useState('')
  const [isRamadan, setIsRamadan] = useState(false)
  const [dailyName, setDailyName] = useState({ arabic:'الرَّحْمَن', english:'Ar-Rahman — The Most Merciful' })
  const [dailyHadith, setDailyHadith] = useState('')
  const particleRef = useRef(null)

  // Fetch prayer times
  useEffect(() => {
    const city = user?.city || 'Delhi'
    const method = user?.madhab === 'shia' ? 0 : 1
    fetch(`https://api.aladhan.com/v1/timingsByCity?city=${city}&country=IN&method=${method}`)
      .then(r => r.json())
      .then(d => {
        if (d.data?.timings) {
          setPrayerTimes(d.data.timings)
          calculateNextPrayer(d.data.timings)
        }
      }).catch(() => {})
  }, [user?.city])

  // Fetch Hijri date
  useEffect(() => {
    const today = new Date()
    const dd = String(today.getDate()).padStart(2,'0')
    const mm = String(today.getMonth()+1).padStart(2,'0')
    const yyyy = today.getFullYear()
    fetch(`https://api.aladhan.com/v1/gToH?date=${dd}-${mm}-${yyyy}`)
      .then(r => r.json())
      .then(d => {
        const h = d.data.hijri
        const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday']
        setHijriDate(`${days[today.getDay()]} · ${h.day} ${h.month.en} ${h.year} AH`)
        // Check Ramadan
        if (h.month.number === 9) setIsRamadan(true)
      }).catch(() => {
        const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday']
        setHijriDate(days[new Date().getDay()])
      })
  }, [])

  // Fetch streak
  useEffect(() => {
    fetch('/api/streak/me', {
      headers: { Authorization: `Bearer ${localStorage.getItem('noor_token')}` }
    }).then(r => r.json())
      .then(d => setStreak({ current: d.currentStreak || 0,
        weekly: d.weeklyCompleted || 0, weeklyGoal: d.weeklyGoal || 5 }))
      .catch(() => {})
  }, [])

  // Countdown timer
  function calculateNextPrayer(times) {
    const prayers = ['Fajr','Dhuhr','Asr','Maghrib','Isha']
    const now = new Date()
    for (const prayer of prayers) {
      if (!times[prayer]) continue
      const [h,m] = times[prayer].split(':').map(Number)
      const pTime = new Date()
      pTime.setHours(h, m, 0, 0)
      if (pTime > now) {
        setNextPrayer({ name: prayer, time: times[prayer] })
        return
      }
    }
    setNextPrayer({ name: 'Fajr', time: times?.Fajr || '04:43' })
  }

  useEffect(() => {
    const interval = setInterval(() => {
      if (!nextPrayer.time) return
      const [h, m] = nextPrayer.time.split(':').map(Number)
      const target = new Date()
      target.setHours(h, m, 0, 0)
      const now = new Date()
      if (target <= now) target.setDate(target.getDate() + 1)
      const diff = Math.max(0, target - now)
      const hh = Math.floor(diff / 3600000)
      const mm2 = Math.floor((diff % 3600000) / 60000)
      const ss = Math.floor((diff % 60000) / 1000)
      setCountdown(`${hh}:${String(mm2).padStart(2,'0')}:${String(ss).padStart(2,'0')}`)

      // Iftar countdown (Maghrib)
      if (prayerTimes?.Maghrib) {
        const [mh, mm3] = prayerTimes.Maghrib.split(':').map(Number)
        const iftar = new Date()
        iftar.setHours(mh, mm3, 0, 0)
        if (iftar <= now) iftar.setDate(iftar.getDate() + 1)
        const idiff = Math.max(0, iftar - now)
        const ih = Math.floor(idiff / 3600000)
        const im = Math.floor((idiff % 3600000) / 60000)
        const is2 = Math.floor((idiff % 60000) / 1000)
        setIftarCountdown(`${ih}:${String(im).padStart(2,'0')}:${String(is2).padStart(2,'0')}`)
      }
    }, 1000)
    return () => clearInterval(interval)
  }, [nextPrayer, prayerTimes])

  const prayers = [
    { key:'Fajr', label:'Fajr', time: prayerTimes?.Fajr || '04:43' },
    { key:'Dhuhr', label:'Dhuhr', time: prayerTimes?.Dhuhr || '12:11' },
    { key:'Asr', label:'Asr', time: prayerTimes?.Asr || '15:58' },
    { key:'Maghrib', label:'Maghrib', time: prayerTimes?.Maghrib || '18:54' },
    { key:'Isha', label:'Isha', time: prayerTimes?.Isha || '20:23' },
  ]

  const weekDots = Array(7).fill(null).map((_, i) => {
    if (i < (streak.weekly - 1)) return 'filled'
    if (i === streak.weekly - 1) return 'today'
    return 'empty'
  })

  const quickSessions = [
    { id:'morning-azkar', icon:'ti-sun', title:'Morning Azkar', meta:'10 min · Arabic + English' },
    { id:'evening-azkar', icon:'ti-moon', title:'Evening Azkar', meta:'10 min · Guided' },
    { id:'dua-anxiety-60', icon:'ti-heart', title:'60s Dua', meta:'1 min · For right now' },
    { id:'quran-al-kahf', icon:'ti-book', title:'Surah Al-Kahf', meta:'14 min · Friday' },
  ]

  return (
    <div style={{ background:'#001a00', minHeight:'100vh', color:'#e8f5e8',
      position:'relative', overflow:'hidden', paddingBottom:80 }}>

      {/* Floating particles */}
      <div style={{ position:'absolute', top:0, left:0, width:'100%',
        height:'100%', pointerEvents:'none', zIndex:0 }}>
        {Array(10).fill(null).map((_, i) => (
          <div key={i} style={{
            position:'absolute',
            left: `${25 + Math.random() * 50}%`,
            bottom: `${5 + Math.random() * 40}%`,
            width: `${1 + Math.random() * 2}px`,
            height: `${1 + Math.random() * 2}px`,
            background: '#ffd700',
            borderRadius: '50%',
            animation: `floatUp ${3 + Math.random() * 4}s ${Math.random() * 5}s linear infinite`,
            opacity: 0,
          }}/>
        ))}
      </div>

      <div style={{ position:'relative', zIndex:1 }}>

        {/* Islamic Banner */}
        {isRamadan && (
          <div style={{ background:'rgba(255,215,0,0.1)',
            borderBottom:'0.5px solid rgba(255,215,0,0.2)',
            padding:'8px 16px', display:'flex',
            justifyContent:'space-between', alignItems:'center' }}>
            <span style={{ fontSize:12, color:'#ffd700' }}>Ramadan Mubarak</span>
            <span style={{ fontFamily:'Amiri,serif', fontSize:14,
              color:'#ffd700', direction:'rtl' }}>رمضان مبارك</span>
          </div>
        )}

        {/* Header */}
        <div style={{ display:'flex', alignItems:'center',
          justifyContent:'space-between', padding:'14px 16px 10px' }}>
          <div style={{ fontSize:22, fontWeight:500, color:'#00a550',
            letterSpacing:4 }}>NOOR</div>
          <div style={{ display:'flex', gap:16, color:'#4a7a4a' }}>
            <i className="ti ti-bell" style={{fontSize:20}} aria-hidden="true"/>
            <i className="ti ti-search" style={{fontSize:20}} aria-hidden="true"/>
            <i className="ti ti-menu-2" style={{fontSize:20}} aria-hidden="true"/>
          </div>
        </div>

        {/* Greeting */}
        <div style={{ padding:'0 16px 14px' }}>
          <div style={{ fontSize:17, fontWeight:500 }}>
            As-salamu alaykum, {user?.name || 'Ahmed'}
          </div>
          <div style={{ fontSize:12, color:'#4a7a4a', marginTop:3 }}>
            {hijriDate || 'Loading...'}
          </div>
        </div>

        {/* Prayer Times Card */}
        <div style={{ margin:'0 16px 14px', background:'rgba(0,165,80,0.08)',
          border:'0.5px solid rgba(0,165,80,0.25)', borderRadius:12, padding:14 }}>
          <div style={{ display:'flex', justifyContent:'space-between',
            alignItems:'center', marginBottom:10 }}>
            <div>
              <div style={{ fontSize:11, color:'#4a7a4a', textTransform:'uppercase',
                letterSpacing:1, marginBottom:2 }}>Next prayer</div>
              <div style={{ fontSize:16, fontWeight:500, color:'#00a550' }}>
                {nextPrayer.name}
              </div>
            </div>
            <div style={{ fontSize:26, fontWeight:500, color:'#ffd700',
              fontVariantNumeric:'tabular-nums' }}>
              {countdown}
            </div>
          </div>
          <div style={{ display:'flex', justifyContent:'space-between',
            paddingTop:10, borderTop:'0.5px solid rgba(0,165,80,0.15)' }}>
            {prayers.map(p => (
              <div key={p.key} style={{ textAlign:'center' }}>
                <div style={{ fontSize:10, color: p.key===nextPrayer.name
                  ? '#00a550' : '#4a7a4a', marginBottom:2 }}>
                  {p.label}{p.key===nextPrayer.name ? ' ▸' : ''}

            width="60" height="60" viewBox="0 0 60 60">
