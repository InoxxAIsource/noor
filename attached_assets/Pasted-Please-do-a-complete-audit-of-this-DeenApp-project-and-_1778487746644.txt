Correct server/index.js structure:
```javascript
import express from 'express'
import cors from 'cors'
import { fileURLToPath } from 'url'
import { dirname, join } from 'path'
import authRoutes from './routes/auth.js'
import prayerRoutes from './routes/prayer.js'
import sessionRoutes from './routes/sessions.js'
import duaRoutes from './routes/duas.js'
import nameRoutes from './routes/names.js'
import streakRoutes from './routes/streak.js'
import progressRoutes from './routes/progress.js'
import aiRoutes from './routes/ai.js'
import masjidRoutes from './routes/masjid.js'
import zakatRoutes from './routes/zakat.js'
import { seedAll } from './seed/index.js'

const app = express()
const __dirname = dirname(fileURLToPath(import.meta.url))

app.use(cors({ origin: '*' }))
app.use(express.json({ limit: '10kb' }))

// API routes
app.use('/api/auth', authRoutes)
app.use('/api/prayer', prayerRoutes)
app.use('/api/sessions', sessionRoutes)
app.use('/api/duas', duaRoutes)
app.use('/api/names', nameRoutes)
app.use('/api/streak', streakRoutes)
app.use('/api/progress', progressRoutes)
app.use('/api/ai', aiRoutes)
app.use('/api/masjid', masjidRoutes)
app.use('/api/zakat', zakatRoutes)

// Serve React in production
const clientDist = join(__dirname, '../client/dist')
app.use(express.static(clientDist))
app.get('*', (req, res) => {
  if (!req.path.startsWith('/api')) {
    res.sendFile(join(clientDist, 'index.html'))
  }
})

const PORT = process.env.PORT || 3000
app.listen(PORT, async () => {
  console.log(`DeenApp server running on port ${PORT}`)
  await seedAll()
})
```

FIX 4 — server/seed/index.js:
Create this file if missing:
```javascript
import Database from '@replit/database'
const db = new Database()
import { sessions } from './sessions.js'
import { duas } from './duas.js'
import { names } from './names.js'
import { namesOfAllah } from './namesOfAllah.js'
import { hadiths } from './hadiths.js'

export async function seedAll() {
  try {
    const existingSessions = await db.get('sessions:all')
    if (!existingSessions) {
      await db.set('sessions:all', JSON.stringify(sessions))
      console.log('✓ Sessions seeded:', sessions.length)
    }
    const existingDuas = await db.get('duas:all')
    if (!existingDuas) {
      await db.set('duas:all', JSON.stringify(duas))
      console.log('✓ Duas seeded:', duas.length)
    }
    const existingNames = await db.get('names:all')
    if (!existingNames) {
      await db.set('names:all', JSON.stringify(names))
      console.log('✓ Names seeded:', names.length)
    }
    const existingAllah = await db.get('namesOfAllah:all')
    if (!existingAllah) {
      await db.set('namesOfAllah:all', JSON.stringify(namesOfAllah))
      console.log('✓ 99 Names seeded')
    }
    const existingHadiths = await db.get('hadiths:all')
    if (!existingHadiths) {
      await db.set('hadiths:all', JSON.stringify(hadiths))
      console.log('✓ Hadiths seeded:', hadiths.length)
    }
  } catch(err) {
    console.error('Seed error:', err.message)
  }
}
```

FIX 7 — Prayer times route:
Ensure server/routes/prayer.js fetches correctly:
```javascript
import express from 'express'
const router = express.Router()

router.get('/times', async (req, res) => {
  try {
    const { city = 'Delhi', lat, lng, madhab = 'sunni' } = req.query
    const method = madhab === 'shia' ? 0 : 1
    let url
    if (lat && lng) {
      url = `https://api.aladhan.com/v1/timings?latitude=${lat}&longitude=${lng}&method=${method}`
    } else {
      url = `https://api.aladhan.com/v1/timingsByCity?city=${encodeURIComponent(city)}&country=IN&method=${method}`
    }
    const response = await fetch(url)
    const data = await response.json()
    res.json(data.data?.timings || {})
  } catch(err) {
    res.status(500).json({ error: err.message })
  }
})

router.get('/hijri', async (req, res) => {
  try {
    const today = new Date()
    const dd = String(today.getDate()).padStart(2,'0')
    const mm = String(today.getMonth()+1).padStart(2,'0')
    const yyyy = today.getFullYear()
    const response = await fetch(`https://api.aladhan.com/v1/gToH?date=${dd}-${mm}-${yyyy}`)
    const data = await response.json()
    res.json(data.data?.hijri || {})
  } catch(err) {
    res.status(500).json({ error: err.message })
  }
})

export default router
```
